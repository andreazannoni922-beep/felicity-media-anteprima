function t( $c, $m ) { global $out; $out[] = ( $c ? 'PASS ' : 'FAIL ' ) . $m; }
function rq( $route, $params, $method = 'POST' ) {
	$r = new WP_REST_Request( $method, '/felicity/v1/' . $route );
	foreach ( $params as $k => $v ) { $r->set_param( $k, $v ); }
	$res = rest_do_request( $r );
	return array( $res->get_status(), $res->get_data() );
}
function page( $view ) { return fm_render( file_get_contents( get_template_directory() . '/views/' . $view . '.html' ), $view ); }
function key_of( $text ) { foreach ( fm_defaults()['t'] as $k => $v ) { if ( $v === $text ) { return $k; } } return null; }

delete_option( FM_OPT ); delete_option( FM_HIST );
$d = fm_defaults();

// visitor: cannot save

foreach ( array( 'index', 'servizi', 'metodo', 'chi-siamo', 'contatti', '404' ) as $v ) {
	$h = page( $v );

		t( is_array( json_decode( $m[1], true ) ), "$v: dati per Google validi" );
	}
}
$h = page( 'index' );
t( false !== strpos( $h, '<span data-fe="' . key_of( 'Le parole giuste,' ) . '">Le parole giuste,</span>' ), 'home: titolo originale con il suo segno' );

t( false !== strpos( page( 'index' ), '>Le parole <b>che servono</b>,alert(1)</span>' ), 'la home mostra il nuovo testo' );
t( isset( $r['cache']['codes'] ), 'svuotamento cache tentato: ' . wp_json_encode( $r['cache'] ) );
list( $st, $r ) = rq( 'modifica', array( 'tipo' => 'testo', 'chiave' => $k, 'valore' => 'Le parole giuste,' ) );
t( ! isset( fm_saved()['t'][ $k ] ), 'rimettere il testo uguale all’originale non lascia modifiche' );
list( $st ) = rq( 'modifica', array( 'tipo' => 'testo', 'chiave' => 'inventata-123', 'valore' => 'x' ) );

// phone: the link follows the text
$kt = null;
foreach ( $d['h'] as $kk => $href ) { if ( 0 === strpos( $href, 'tel:' ) && 0 === strpos( $kk, 'index-' ) ) { $kt = $kk; } }
rq( 'modifica', array( 'tipo' => 'testo', 'chiave' => $kt, 'valore' => '333 1234567' ) );

$kf = null;
foreach ( $d['t'] as $kk => $v ) { if ( 0 === strpos( $kk, 'comune-' ) && false !== strpos( $v, 'P.IVA' ) ) { $kf = $kk; } }
